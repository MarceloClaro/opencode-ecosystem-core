% fontspec/babel, e ANTES de \begin{document}.
%
% Engine: XeLaTeX + Biber (nunca pdflatex).
%
% Caminho dos componentes: por padrão procura os design-*.tex no diretório
% atual / input path. Para apontar outra pasta, antes de carregar este arquivo:
%   \newcommand{\imersivocfg}{caminho/para/abntex2-imersivo/}

\providecommand{\imersivocfg}{}

% --- Pacotes exigidos pela camada imersiva ---------------------------------
\usepackage{tikz}
\usetikzlibrary{shapes, arrows, positioning, calc, decorations.pathreplacing}
\usepackage[export]{adjustbox}   % \resizebox p/ figuras largas no miolo estreito
\usepackage{lettrine}            % letra capitular (\capitular)
\usepackage[most]{tcolorbox}     % caixas pedagógicas
\usepackage{fontawesome5}        % ícones
\usepackage{qrcode}              % elementos digitais (\colab/\repo/\dataset)

% --- Compatibilidade sidenotes x abntex2/memoir ----------------------------
% sidenotes exige changepage[strict]; abntex2/memoir "emula" changepage sem
% opções e predefine \sidecaption/marginfigure/margintable. Resolvemos o
% choque ANTES de carregar sidenotes (nenhum desses comandos do memoir é
% usado; sidenotes assume o papel deles).
\PassOptionsToPackage{strict}{changepage}
\makeatletter
\let\sidecaption\undefined
\let\marginfigure\undefined
\let\endmarginfigure\undefined
\let\margintable\undefined
\let\endmargintable\undefined
\makeatother
\usepackage{sidenotes}           % puxa marginnote
