\label{fig:performance}
\end{figure}

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/fig2_fairness_heatmap.png}
\caption{Fairness Metrics Heatmap. (A) Ethnicity-based fairness metrics, 
(B) Gender-based fairness metrics. Color intensity indicates metric 
magnitude (darker = higher bias). EOD: Equalized Odds Difference, 
DPD: Demographic Parity Difference, FPR: False Positive Rate.}
\label{fig:heatmap}
\end{figure}

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/fig3_intersectional.png}
\caption{Intersectional Analysis (Gender $\times$ Ethnicity). 
(A) Accuracy by subgroup, (B) F1-Score by subgroup, (C) True Positive 
Rate by subgroup. Minority-Female group shows lowest performance across 
all metrics, indicating amplified bias at the intersection.}
\label{fig:intersectional}
\end{figure}

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/fig4_fpr_disparity.png}
\caption{False Positive Rate Disparity by Ethnicity. Red bars (Minority) 
consistently exceed blue bars (Majority) across all models, indicating 
systematic over-diagnosis bias. Gap values ($\Delta$) shown above each 
model.}
\label{fig:fpr_disparity}
\end{figure}
