\section{Prancha 1: um mapa mental do trabalho}

\begin{flowch}[Perguntas que cercam a nota científica]{ligações: organização conceitual, sem ordem obrigatória}
\node[fne,text width=3.4cm] (centro) {Nota científica\\com evidências};
\node[fns,text width=3.2cm,above left=13mm and 14mm of centro] (pergunta) {Qual é a pergunta?\\Fonte e escopo};
\node[fn,text width=3.2cm,above right=13mm and 14mm of centro] (metodo) {Como examinar?\\Premissas e método};
\node[fn,text width=3.2cm,below left=13mm and 14mm of centro] (resultado) {O que ocorreu?\\Artefatos e recibos};
\node[fns,text width=3.2cm,below right=13mm and 14mm of centro] (limite) {O que concluir?\\Aceite e limitações};
\draw[seta] (centro)--(pergunta);
\draw[seta] (centro)--(metodo);
\draw[seta] (centro)--(resultado);
\draw[seta] (centro)--(limite);
\end{flowch}
\begin{descricao}
\textbf{Como funciona e por quê. Mecanismo.} As ligações distribuem quatro perguntas
sobre o mesmo artefato. A fonte informa o problema; o método orienta a operação;
o recibo registra o que foi observado; o aceite compara esse resultado ao contrato.
\textbf{Justificativa.} A disposição permite localizar uma lacuna sem confundir
método planejado com resultado produzido.
\textbf{Limite.} É um mapa mental editorial, não o grafo de chamadas do software.
\end{descricao}

\section{Prancha 2: a encomenda e os seus estados}

\begin{flowch}[Receber não é entregar]{setas: transições que exigem evidência própria}
\matrix[matrix of nodes,name=estado,row sep=3.5mm,column sep=5mm,nodes={fn},ampersand replacement=\&]{
 |[fns]|Pedido delimitado \\
 |[fn]|Tarefa atribuída \\
 |[fn]|Operação executada \\
 |[fne]|Entrega conferida \\
};
\draw[seta] (estado-1-1)--(estado-2-1);
\draw[seta] (estado-2-1)--node[right,font=\scriptsize]{requer executor} (estado-3-1);
\draw[seta] (estado-3-1)--node[right,font=\scriptsize]{requer gate} (estado-4-1);
\end{flowch}
\begin{descricao}
\textbf{Como funciona e por quê. Mecanismo.} A atribuição dá uma identidade à tarefa.
A seta seguinte depende de um executor que realmente a consuma. Depois, um gate
pode avaliar a entrega conforme seus critérios. Um retorno \cod{pending_executor}
não percorreu automaticamente as duas últimas caixas.
\textbf{Justificativa.} O rastro de estados impede que uma mensagem de encaminhamento
seja apresentada como cálculo ou análise concluída.
\textbf{Limite.} A aceitação computacional local continua separada da revisão científica.
\end{descricao}

\section{Prancha 3: o caderno e a próxima pergunta}

\begin{flowch}[Memória como registro examinável]{setas: gravação e recuperação; sem autopromoção de confiança}
\matrix[matrix of nodes,name=cad,row sep=4mm,column sep=4mm,nodes={fn},ampersand replacement=\&]{
 |[fns]|Resultado ou falha \& |[fn]|Observação tipada \\
 |[fn]|Novo briefing \& |[fn]|Persistência e origem \\
};
\draw[seta] (cad-1-1)--(cad-1-2);
\draw[seta] (cad-1-2)--(cad-2-2);
\draw[seta] (cad-2-2)--(cad-2-1);
\end{flowch}
\begin{descricao}
\textbf{Como funciona e por quê. Mecanismo.} Uma observação conserva resultado,
origem, limites e próxima ação. Uma consulta posterior pode recuperá-la para o
briefing. As setas representam armazenamento e consulta, não aumento automático
de competência ou do ledger de confiança.
\textbf{Justificativa.} Separar memória e julgamento permite revisar uma observação
sem transformá-la em verdade apenas porque foi persistida.
\textbf{Limite.} Recuperação lexical e relevância heurística podem falhar ou trazer
contexto insuficiente; a aplicação precisa declarar a degradação observada.
\end{descricao}

\section{Prancha 4: a porta de conferência}

\begin{flowch}[O fiscal confere um contrato]{setas: percurso de avaliação com ramo de recusa}
\node[fns] (artefato) {Spec, artefato e\\evidência de teste};
\node[fde,below=8mm of artefato,text width=2.8cm] (gate) {Critérios e custódia\\atendidos?};
\node[fns,below left=8mm and 9mm of gate] (recusa) {Recusar com motivo};
\node[fne,below right=8mm and 9mm of gate] (aceite) {Aceitar no escopo};
\draw[seta] (artefato)--(gate);
\draw[seta] (gate)--node[above left,font=\scriptsize]{não} (recusa);
\draw[seta] (gate)--node[above right,font=\scriptsize]{sim} (aceite);
\end{flowch}
\begin{descricao}
\textbf{Como funciona e por quê. Mecanismo.} O gate confronta a entrega com as
condições da especificação e com as evidências exigidas. A recusa conserva um
motivo; o ramo positivo vale para aquele contrato.
\textbf{Justificativa.} Uma decisão explícita torna examinável por que o fluxo
avançou ou parou, sem usar uma nota média para compensar requisito obrigatório.
\textbf{Limite.} O desenho não demonstra correção universal, mérito editorial ou
validação empírica fora dos casos conferidos.
\end{descricao}

\section{Prancha 5: o que fica na máquina e o que atravessa a fronteira}

\begin{flowch}[Uma ferramenta, dois percursos]{setas: chamadas condicionais; ler a política de cada operação}
\matrix[matrix of nodes,name=fronteira,row sep=5mm,column sep=7mm,nodes={fn},ampersand replacement=\&]{
 |[fne]|Primário \& |[fn]|Operação local \\
 |[fn]|Ticket permitido \& |[fn]|Host e serviço externo \\
 |[fns]|Resposta vinculada \& |[fn]|Recibo e limites \\
};
\draw[seta] (fronteira-1-1)--(fronteira-1-2);
\draw[seta] (fronteira-1-1)--(fronteira-2-1);
\draw[seta] (fronteira-2-1)--(fronteira-2-2);
\draw[seta] (fronteira-2-2)--(fronteira-3-1);
\draw[seta] (fronteira-3-1)--(fronteira-3-2);
\end{flowch}
\begin{descricao}
\textbf{Como funciona e por quê. Mecanismo.} Uma operação local e uma operação
hospedada não têm a mesma fronteira de dados. Na rota de plugins hospedados, o
primário prepara o ticket permitido e registra a resposta vinculada.
\textbf{Justificativa.} A localização do arquivo do plugin não informa onde a
ferramenta executa. A fronteira precisa ser examinada por operação.
\textbf{Limite.} Uma amostra Wolfram hospedada não demonstra kernel licenciado local
nem execução offline. O REA isolado também não controla um cliente LLM externo.
\end{descricao}

\begin{flowch}[Antes de reutilizar uma métrica]{ligações: dimensões de interpretação, não graus de mérito}

\draw[seta] (numero-1-2)--(numero-2-2);
\draw[seta] (numero-2-2)--(numero-2-1);
\end{flowch}
\begin{descricao}
\textbf{Como funciona e por quê. Mecanismo.} Para interpretar um resultado, identifique

\textbf{Justificativa.} Essas perguntas evitam comparar agentes, categorias, arquivos
e testes como se fossem a mesma unidade.
\textbf{Limite.} Uma seleção sem falhas observadas não garante ausência de bugs nem
produz validação científica externa.
\end{descricao}

\section[Leitura guiada]{Leitura guiada: usar a figura sem promover seu conteúdo a prova}
\begin{descricao}
\textbf{O fenômeno.} Uma ilustração facilita a orientação, mas números e setas podem resumir de modo impreciso um mecanismo ou uma medição histórica.
\end{descricao}
\begin{definicao}
\textbf{Grandezas e definições.} Diferencie figura didática, fonte local e recibo de execução. A quantidade de figuras descreve o atlas; não é contagem de experimentos ou validações.
\end{definicao}
\begin{metodo}
\textbf{Dedução passo a passo.} Escolha uma figura; leia sua legenda; localize o capítulo associado; confira cada número na fonte indicada; marque o estado temporal da informação; revise antes de apresentar.
\end{metodo}
\begin{resultado}
\textbf{Exemplo resolvido.} Ana encontra um número decorativo no infográfico que difere do inventário corrente. Ela utiliza o inventário como referência operacional e explica que a figura é uma representação didática; não apresenta o número decorativo como nova medição.
\end{resultado}
\begin{resultado}
\textbf{Ordem de grandeza e verificação.} Este conjunto de pranchas analógicas contém seis figuras declaradas; ele complementa as 54 pranchas de arquitetura do atlas R911. Conte os arquivos e confira legibilidade, correspondência com a legenda e limites de cada uma. Se uma figura falhar, registre o problema individualmente.
\end{resultado}
\begin{aviso}
\textbf{Limite de validade.} A inspeção visual registrada em 11/10 tem o alcance descrito na legenda. Ela não calibra métricas nem confirma eficácia do Core. Este capítulo não transforma imagens geradas por IA em dados experimentais.
\end{aviso}
\begin{resultado}
\textbf{Exercícios.} Relacione uma figura a uma fonte; encontre um número que exige data e escopo; descreva uma simplificação do desenho; escreva uma legenda que conserve esse limite.
\end{resultado}
